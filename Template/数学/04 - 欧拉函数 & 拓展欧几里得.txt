i64 phi(i64 x) {
	i64 res = x;
	for (int i = 2; i <= x / i; i ++) {
		if (x % i == 0) {
			res = res / i * (i - 1);
		}
		while (x % i == 0) {
			x /= i;
		}
	}
	if (x > 1) {
		res = res / x * (x - 1);
	}
	return res;
}

int exgcd(int a, int b, int& x, int& y) {
	if (!b) {
		x = 1, y = 0;
		return a;
	}
	int d = exgcd(b, a % b, x, y);
	int X = x;
	x = y, y = X - (a / b) * y;
	return d;
}
