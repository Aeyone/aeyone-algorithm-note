int unit = 990; // 块大小
int getBlock(int x){
	return x / unit;
}
struct Query {
  	int l, r, id;
  	bool operator<(const Query &x) const { // 奇偶排序
	  	int a = getBlock(l), b = getBlock(x.l);
	    if (a != b){
	    	return l < x.l;
	    }
	    if(a & 1){
	    	return r < x.r;
	    }else{
		    return r > x.r;
	    }
  	}
};

void solve() {
	int q;
	cin >> q;
	vector<Query> que(q);
	for(int i = 0; i < q; i ++){
		cin >> que[i].l >> que[i].r;
		que[i].id = i;
	}
	sort(que.begin(), que.end());
	int l = 1, r = 0, res = 0; // 左闭右闭

	auto add = [&](int u)->void{
	};
	auto del = [&](int u)->void{
	};
	
	for(register int i = 0; i < q; i ++){
		auto &[L, R, id] = que[i];
		while(l > L) add(-- l);
		while(r < R) add(++ r);
		while(l < L) del(l ++);
		while(r > R) del(r --);
		ans[id] = res;
	}
}
